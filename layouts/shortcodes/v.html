{{- /* 圣经经文引用 Shortcode 使用方式： {{< v "撒上7:12" >}} {{< v "约3:16-17" >}} {{< v "罗8:28 30-31" >}} {{< v "诗23:1-3 5" >}} */ -}} {{- $reference := strings.TrimSpace (.Get 0) -}} {{- $parts := split $reference ":" -}} {{- if ne (len $parts) 2 -}} {{- errorf "经文引用格式错误：%s，正确格式：{{< v \"罗8:28 30-31\" >}}" $reference -}} {{- end -}} {{- $head := index $parts 0 -}} {{- $verses := strings.TrimSpace (index $parts 1) -}} {{- $book := replaceRE "[0-9]+$" "" $head -}} {{- $chapter := strings.TrimPrefix $book $head -}} {{- if or (eq $book "") (not (findRE "^[0-9]+$" $chapter)) -}} {{- errorf "书卷或章节格式错误：%s" $reference -}} {{- end -}} {{- if not (findRE "^[0-9]+(-[0-9]+)?( [0-9]+(-[0-9]+)?)*$" $verses) -}} {{- errorf "经文节数格式错误：%s" $reference -}} {{- end -}} {{- $id := printf "bible-verse-%d" .Ordinal -}} {{- /* 每页只加载一次 CSS 和 JS */ -}} {{- if not (.Page.Store.Get "bibleVerseAssets") -}} {{- .Page.Store.Set "bibleVerseAssets" true -}} {{- end -}}
